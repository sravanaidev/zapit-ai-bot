  ZAPIT — REMOVE AND INSTALL REFERENCE GUIDE

════════════════════════════════════════════════════════
  STEP 1 — REMOVE OLD VERSION
════════════════════════════════════════════════════════

── SERVER B (run first) ────────────────────────────────

sudo systemctl stop zapit-recv
sudo systemctl disable zapit-recv
sudo rm -f /usr/local/bin/zapit-send
sudo rm -f /usr/local/bin/zapit-recv
sudo rm -f /etc/systemd/system/zapit-recv.service
sudo systemctl daemon-reload
rm -rf ~/zapit
rm -f /var/lib/zapit/manifests/*.manifest.json
rm -f /var/run/zapit/*.pid
rm -f /data/received/.*.zapit-tmp

── SERVER A ────────────────────────────────────────────

sudo rm -f /usr/local/bin/zapit-send
sudo rm -f /usr/local/bin/zapit-recv
rm -rf ~/zapit
rm -f /var/lib/zapit/manifests/*.manifest.json
rm -f /var/run/zapit/*.pid

── VERIFY CLEAN (both servers) ─────────────────────────

which zapit-send
which zapit-recv
ls ~/zapit 2>/dev/null || echo "folder gone"

════════════════════════════════════════════════════════
  STEP 3 — BUILD (both servers)
════════════════════════════════════════════════════════

cd zapit
go mod tidy
make build

── Expected output ─────────────────────────────────────

════════════════════════════════════════════════════════
  STEP 4 — INSTALL BINARIES (both servers)
════════════════════════════════════════════════════════

sudo cp bin/zapit-send /usr/local/bin/
sudo cp bin/zapit-recv /usr/local/bin/
sudo chmod +x /usr/local/bin/zapit-send
sudo chmod +x /usr/local/bin/zapit-recv


════════════════════════════════════════════════════════
  STEP 5 — CREATE DIRECTORIES (both servers)
════════════════════════════════════════════════════════

sudo mkdir -p /var/lib/zapit/manifests
sudo mkdir -p /var/log/zapit
sudo mkdir -p /var/run/zapit
sudo chown $(whoami):$(whoami) /var/lib/zapit
sudo chown $(whoami):$(whoami) /var/lib/zapit/manifests
sudo chown $(whoami):$(whoami) /var/log/zapit
sudo chown $(whoami):$(whoami) /var/run/zapit


════════════════════════════════════════════════════════
  STEP 6 — INSTALL SERVICE (Server B only)
════════════════════════════════════════════════════════

sudo tee /etc/systemd/system/zapit-recv.service > /dev/null << 'EOF'
[Unit]
Description=Zapit receiver daemon
After=network-online.target
Wants=network-online.target

[Service]
Type=simple
User=ec2-user
ExecStart=/usr/local/bin/zapit-recv serve \
  --listen          0.0.0.0:2222 \
  --host-key        /etc/zapit/host_key \
  --authorized-keys /etc/zapit/authorized_keys
Restart=always
RestartSec=3s
TimeoutStartSec=30
TimeoutStopSec=60
StandardOutput=journal
StandardError=journal
SyslogIdentifier=zapit-recv
LimitNOFILE=65536

[Install]
WantedBy=multi-user.target
EOF

sudo systemctl daemon-reload
sudo systemctl enable zapit-recv
sudo systemctl start zapit-recv
