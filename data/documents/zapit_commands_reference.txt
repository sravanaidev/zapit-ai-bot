================================================================================
1. SERVER SETUP (Fresh EC2 Instance)
================================================================================

# Install prerequisites
sudo yum install -y git make rpm-build

# Install Go
wget https://go.dev/dl/go1.22.4.linux-amd64.tar.gz
sudo tar -C /usr/local -xzf go1.22.4.linux-amd64.tar.gz
echo 'export PATH=$PATH:/usr/local/go/bin' >> ~/.bashrc
source ~/.bashrc
go version

# Install garble (obfuscation)
go install mvdan.cc/garble@v0.12.1
export PATH=$PATH:$(go env GOPATH)/bin

# Clone repo
cd ~
git clone https://github.com/saisharan5111/zapit.3.0.git zapit-4.4
cd zapit-4.4

# Build
go mod tidy
make build

================================================================================
2. SERVER A (SENDER) SETUP
================================================================================

# Create directories
sudo mkdir -p /etc/zapit
sudo mkdir -p /var/lib/zapit/manifests
sudo mkdir -p /var/log/zapit
sudo mkdir -p /var/run/zapit
sudo chown -R ec2-user:ec2-user /var/lib/zapit /var/log/zapit /var/run/zapit

# Generate SSH keys
ssh-keygen -t ed25519 -f ~/.ssh/zapit_key -N "" -C "zapit-transfer-key"
ssh-keygen -t ed25519 -f /etc/zapit/host_key -N "" -C "zapit-host-key"
sudo chown ec2-user:ec2-user /etc/zapit/host_key
cp ~/.ssh/zapit_key.pub /etc/zapit/authorized_keys

# Create nodes.conf (replace SERVER_B_IP)
cat > /etc/zapit/nodes.conf << 'EOF'
[EC2_SERVER_B]
host        = SERVER_B_IP
port        = 2222
user        = ec2-user
key         = /home/ec2-user/.ssh/zapit_key
known_hosts = /etc/zapit/known_hosts
insecure    = true
encrypt     = true
init_window = 64
max_window  = 512
EOF

# UDP buffer tuning
sudo tee /etc/sysctl.d/99-zapit.conf << 'EOF'
net.core.rmem_max=268435456
net.core.wmem_max=268435456
net.core.rmem_default=26214400
net.core.wmem_default=26214400
net.core.netdev_max_backlog=50000
EOF
sudo sysctl -p /etc/sysctl.d/99-zapit.conf

# Install binaries
sudo cp bin/zapitd       /usr/local/bin/
sudo cp bin/zapit        /usr/local/bin/
sudo cp bin/zapit-send   /usr/local/bin/
sudo cp bin/zapit-reader /usr/local/bin/

# Set zapit-reader setuid
sudo groupadd zapit 2>/dev/null || true
sudo usermod -aG zapit ec2-user
sudo chown root:zapit /usr/local/bin/zapit-reader
sudo chmod 4750 /usr/local/bin/zapit-reader

# Create systemd service
sudo tee /etc/systemd/system/zapitd.service << 'EOF'
[Unit]
Description=Zapit daemon
After=network.target
[Service]
Type=simple
User=ec2-user
ExecStart=/usr/local/bin/zapitd serve \
  --listen 0.0.0.0:2222 \
  --host-key /etc/zapit/host_key \
  --authorized-keys /etc/zapit/authorized_keys \
  --nodes-conf /etc/zapit/nodes.conf \
  --log-dir /var/log/zapit \
  --manifest-dir /var/lib/zapit/manifests \
  --max-concurrent 4
Restart=on-failure
RestartSec=3
LimitNOFILE=65536
[Install]
WantedBy=multi-user.target
EOF

sudo systemctl daemon-reload
sudo systemctl enable zapitd
sudo systemctl start zapitd

# Record Server B host key
ssh-keyscan -p 2222 SERVER_B_IP >> /etc/zapit/known_hosts 2>/dev/null

# Create test files
dd if=/dev/urandom of=/home/ec2-user/testfile_15gb.bin bs=1M count=15360 status=progress

================================================================================
4. TRANSFER COMMANDS
================================================================================

# Basic transfer (AIMD default)
zapit submit \
  --src /home/ec2-user/testfile_15gb.bin \
  --dest /data/received/file.bin \
  --node EC2_SERVER_B \
  --remote-user ec2-user

# SPRINT+ (no target)
zapit submit \
  --src /home/ec2-user/testfile_15gb.bin \
  --dest /data/received/test_sprint.bin \
  --node EC2_SERVER_B --remote-user ec2-user \
  --congestion sprint

# SPRINT+ Dedicated (500 MB/s target)
zapit submit \
  --src /home/ec2-user/testfile_15gb.bin \
  --dest /data/received/test_dedicated.bin \
  --node EC2_SERVER_B --remote-user ec2-user \
  --congestion sprint --target-rate 500m

# With SHA-256 verification
zapit submit \
  --src /home/ec2-user/testfile_15gb.bin \
  --dest /data/received/test_verified.bin \
  --node EC2_SERVER_B --remote-user ec2-user \
  --congestion sprint --target-rate 500m --verify

# With rate limit
zapit submit \
  --src /data/file.bin \
  --dest /data/received/file.bin \
  --node SERVER_B --remote-user ec2-user \
  --rate-limit 100m

# With source user
zapit submit \
  --src /app/data/export.csv \
  --dest /data/incoming/export.csv \
  --node SERVER_B --remote-user ec2-user \
  --source-user app_user

# 1 Gbps target
zapit submit \
  --src /data/large.bin --dest /data/received/large.bin \
  --node SERVER_B --remote-user ec2-user \
  --congestion sprint --target-rate 1g

# Find errors

# Check verification results
grep "CHECKSUM\|verification\|SHA-256\|AES-256" /var/log/zapit/zapit-zapitd-$(date +%Y-%m-%d).log | tail -10

grep "commit\|Commit\|committed" /var/log/zapit/zapit-zapitd-$(date +%Y-%m-%d).log | tail -5
